{{/* ============================================================ agenda-calendar-links.html — context: pàgina d'agenda (RegularPage) Genera un fitxer .ics (Apple Calendar, Google, Outlook…) i el retorna com una icona de calendari per posar al costat de la data. ============================================================ */}} {{- $d := .Params.date_start -}} {{- if $d -}} {{- $t1 := .Params.time_start | default "10:00" -}} {{- $t2 := .Params.time_end | default $t1 -}} {{- $yyyymmdd := replace $d "-" "" -}} {{- $dtStart := printf "%sT%s00" $yyyymmdd (replace $t1 ":" "") -}} {{- $dtEnd := printf "%sT%s00" $yyyymmdd (replace $t2 ":" "") -}} {{- $loc := .Params.location | default .Params.lloc | default "Llumàtics — Nau Bostik, La Sagrera, Barcelona" -}} {{- $title := .Title -}} {{- $desc := .Params.description | default .Params.organizer | default $title -}} {{- $titleEsc := replace (replace $title "," "\\,") ";" "\\;" -}} {{- $descEsc := replace (replace $desc "," "\\,") ";" "\\;" -}} {{- $locEsc := replace (replace $loc "," "\\,") ";" "\\;" -}} {{- $uid := printf "%s-%s@llumatics.com" .File.ContentBaseName $yyyymmdd -}} {{- $icsLines := slice "BEGIN:VCALENDAR" "VERSION:2.0" "PRODID:-//Llumatics//Agenda//CA" "CALSCALE:GREGORIAN" "BEGIN:VEVENT" (printf "UID:%s" $uid) (printf "DTSTAMP:%s" (now.Format "20060102T150405Z")) (printf "DTSTART:%s" $dtStart) (printf "DTEND:%s" $dtEnd) (printf "SUMMARY:%s" $titleEsc) (printf "LOCATION:%s" $locEsc) (printf "DESCRIPTION:%s" $descEsc) "END:VEVENT" "END:VCALENDAR" "" -}} {{- $ics := delimit $icsLines "\r\n" -}} {{- $res := resources.FromString (printf "calendar/%s-%s.ics" .File.ContentBaseName .Language.Lang) $ics -}} {{- end -}}
